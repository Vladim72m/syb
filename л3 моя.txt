CREATE DATABASE ll
USE ll

CREATE TABLE faculty (
    id INT PRIMARY KEY IDENTITY(1,1),
    faculty_name NVARCHAR(200) NOT NULL
);

CREATE TABLE form (
    id INT PRIMARY KEY IDENTITY(1,1),
    form_name NVARCHAR(200) NOT NULL
);

CREATE TABLE hours (
    id INT PRIMARY KEY IDENTITY(1,1),
    course INT NOT NULL,
    faculty_id INT NOT NULL,
    form_id INT NOT NULL,
    all_h INT,
    inclass_h INT,
    CONSTRAINT Fhours_faculty FOREIGN KEY (faculty_id) REFERENCES faculty(id),
    CONSTRAINT hours_form FOREIGN KEY (form_id) REFERENCES form(id)
);

CREATE TABLE stud (
    id INT PRIMARY KEY IDENTITY(1,1),
    last_name NVARCHAR(200) NOT NULL,
    f_name NVARCHAR(200) NOT NULL,
    s_name NVARCHAR(200),
    br_date DATE,
    in_date DATE,
    exm FLOAT
);

CREATE TABLE process (
    stud_id INT NOT NULL,
    hours_id INT NOT NULL,
    CONSTRAINT stud_hours PRIMARY KEY (stud_id, hours_id),
    CONSTRAINT process_stud FOREIGN KEY (stud_id) REFERENCES stud(id),
    CONSTRAINT process_hours FOREIGN KEY (hours_id) REFERENCES hours(id)
);

SELECT stud.last_name, stud.f_name, stud.s_name, faculty.faculty_name, hours.course FROM stud
JOIN process ON stud.id = process.stud_id
JOIN hours ON process.hours_id = hours.id
JOIN faculty ON hours.faculty_id = faculty.id
JOIN form ON hours.form_id = form.id
WHERE form.form_name = 'заочная'
AND DATEDIFF(year, stud.br_date, GETDATE()) < 37;


SELECT faculty.faculty_name, COUNT(stud.id) AS stud_name FROM faculty
JOIN hours ON faculty.id = hours.faculty_id
JOIN process ON hours.id = process.hours_id
JOIN stud ON process.stud_id = stud.id
GROUP BY faculty.faculty_name;

SELECT form.form_name, COUNT(stud.id) AS stud_form FROM form
JOIN hours ON form.id = hours.form_id
JOIN process ON hours.id = process.hours_id
JOIN stud ON process.stud_id = stud.id
GROUP BY form.form_name;

SELECT faculty.faculty_name, AVG(DATEDIFF(year, stud.br_date, GETDATE())) AS faculty_br FROM faculty
JOIN hours ON faculty.id = hours.faculty_id
JOIN process ON hours.id = process.hours_id
JOIN stud ON process.stud_id = stud.id
GROUP BY faculty.faculty_name;

FROM stud
