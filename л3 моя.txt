CREATE DATABASE labb3;
USE labb3;

CREATE TABLE faculty (
    id INT PRIMARY KEY IDENTITY(1,1),
    name NVARCHAR(50) NOT NULL
);

CREATE TABLE education_form (
    id INT PRIMARY KEY IDENTITY(1,1),
    name NVARCHAR(50) NOT NULL
);

CREATE TABLE academic_load (
    id INT PRIMARY KEY IDENTITY(1,1),
    course_number INT NOT NULL,
    faculty_id INT NOT NULL,
    form_id INT NOT NULL,
    total_hours INT,
    classroom_hours INT,
    CONSTRAINT FK_academic_load_faculty FOREIGN KEY (faculty_id) REFERENCES faculty(id),
    CONSTRAINT FK_academic_load_form FOREIGN KEY (form_id) REFERENCES education_form(id)
);

CREATE TABLE student (
    id INT PRIMARY KEY IDENTITY(1,1),
    last_name NVARCHAR(25) NOT NULL,
    first_name NVARCHAR(25) NOT NULL,
    middle_name NVARCHAR(25),
    birth_date DATE,
    enrollment_date DATE,
    exam_score FLOAT
);

CREATE TABLE enrollment (
    student_id INT NOT NULL,
    load_id INT NOT NULL,
    CONSTRAINT PK_enrollment PRIMARY KEY (student_id, load_id),
    CONSTRAINT FK_enrollment_student FOREIGN KEY (student_id) REFERENCES student(id),
    CONSTRAINT FK_enrollment_load FOREIGN KEY (load_id) REFERENCES academic_load(id)
);

-- Запрос 1: Студенты вечерней формы младше 37 лет
SELECT student.last_name, student.first_name, student.middle_name, faculty.name, academic_load.course_number
FROM student
JOIN enrollment ON student.id = enrollment.student_id
JOIN academic_load ON enrollment.load_id = academic_load.id
JOIN faculty ON academic_load.faculty_id = faculty.id
JOIN education_form ON academic_load.form_id = education_form.id
WHERE education_form.name = 'вечерняя'
AND DATEDIFF(year, student.birth_date, GETDATE()) < 37;

-- Запрос 2: Количество студентов по факультетам
SELECT faculty.name, COUNT(student.id) AS student_count
FROM faculty
JOIN academic_load ON faculty.id = academic_load.faculty_id
JOIN enrollment ON academic_load.id = enrollment.load_id
JOIN student ON enrollment.student_id = student.id
GROUP BY faculty.name;

-- Запрос 3: Количество студентов по формам обучения
SELECT education_form.name, COUNT(student.id) AS student_count
FROM education_form
JOIN academic_load ON education_form.id = academic_load.form_id
JOIN enrollment ON academic_load.id = enrollment.load_id
JOIN student ON enrollment.student_id = student.id
GROUP BY education_form.name;

-- Запрос 4: Средний возраст студентов по факультетам
SELECT faculty.name, AVG(DATEDIFF(year, student.birth_date, GETDATE())) AS avg_age
FROM faculty
JOIN academic_load ON faculty.id = academic_load.faculty_id
JOIN enrollment ON academic_load.id = enrollment.load_id
JOIN student ON enrollment.student_id = student.id
GROUP BY faculty.name;

-- Запрос 5: Студенты без отчества
SELECT student.last_name, student.first_name, student.enrollment_date, faculty.name, academic_load.course_number, education_form.name
FROM student
JOIN enrollment ON student.id = enrollment.student_id
JOIN academic_load ON enrollment.load_id = academic_load.id
JOIN faculty ON academic_load.faculty_id = faculty.id
JOIN education_form ON academic_load.form_id = education_form.id
WHERE student.middle_name IS NULL;

-- Запрос 6: Студенты, поступившие в 2015 году (данных нет, результат будет пустым)
SELECT faculty.name, COUNT(student.id) AS cnt
FROM student
JOIN enrollment ON student.id = enrollment.student_id
JOIN academic_load ON enrollment.load_id = academic_load.id
JOIN faculty ON academic_load.faculty_id = faculty.id
WHERE YEAR(student.enrollment_date) = 2015
GROUP BY faculty.name
ORDER BY cnt DESC;

-- Запрос 7: Студенты 2014 года (данных нет)

-- Запрос 8: Факультеты, где есть вечерняя форма
SELECT DISTINCT faculty.name
FROM faculty
JOIN academic_load ON faculty.id = academic_load.faculty_id
JOIN education_form ON academic_load.form_id = education_form.id
WHERE education_form.name = 'вечерняя';

-- Запрос 9: Все комбинации факультетов, форм и курсов
SELECT faculty.name, education_form.name, academic_load.course_number
FROM faculty
JOIN academic_load ON faculty.id = academic_load.faculty_id
JOIN education_form ON academic_load.form_id = education_form.id
ORDER BY faculty.name, education_form.name, academic_load.course_number;

-- Запрос 10: Количество студентов по факультетам и формам
SELECT faculty.name, education_form.name, COUNT(student.id) AS cnt
FROM faculty
JOIN academic_load ON faculty.id = academic_load.faculty_id
JOIN education_form ON academic_load.form_id = education_form.id
JOIN enrollment ON academic_load.id = enrollment.load_id
JOIN student ON enrollment.student_id = student.id
GROUP BY faculty.name, education_form.name
ORDER BY faculty.name;

-- Запрос 11: Студенты на курсах 1 или 3
